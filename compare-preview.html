SQL 预览
下游预览节点将作为子查询消费此结果
LIVE-- SRC = orders_yesterday · TGT = orders_today
WITH
  src AS (SELECT id, user_id, amount FROM orders_yesterday),
  tgt AS (SELECT id, user_id, amount FROM orders_today)
SELECT
  CASE
    WHEN src.id IS NULL              THEN '+'  -- TGT 新增
    WHEN tgt.id IS NULL              THEN '-'  -- SRC 存在但 TGT 删除
    WHEN src.user_id <> tgt.user_id
      OR src.amount   <> tgt.amount THEN '~'  -- 修改
    ELSE                                          '='
  END AS cmp_op,
  COALESCE(src.id, tgt.id) AS cmp_id,
  src.*, tgt.*
FROM src FULL OUTER JOIN tgt ON src.id = tgt.id
WHERE cmp_op IN ('+', '-', '~')
